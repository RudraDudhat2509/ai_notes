WHY JWT EXISTS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Basic auth (password every request) → password leaks on every call
  Sessions → need server-side storage, doesn't scale horizontally
  JWT → self-contained, server stores nothing, no DB lookup per request


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
JWT STRUCTURE
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  header.payload.signature   (three base64 strings joined by dots)

  Header:    {"alg": "HS256", "typ": "JWT"}
  Payload:   {"user_id": 123, "role": "admin", "exp": 1700000000}
  Signature: HMAC_SHA256(base64(header) + "." + base64(payload), SECRET_KEY)

  JWT is SIGNED not ENCRYPTED.
  Anyone can decode the payload — they just can't tamper with it.
  NEVER put passwords, credit cards, or secrets in the payload.

HOW VERIFICATION WORKS (no DB needed)
--------------------------------------
  1. Re-compute signature from header + payload using SECRET_KEY
  2. If matches → payload is untampered → read user_id directly
  3. If mismatch → rejected

  Tamper with payload → signature won't match → rejected automatically.

ACCESS TOKEN + REFRESH TOKEN PATTERN
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Access token  — short-lived (15 min), used for every API call, NOT stored
  Refresh token — long-lived (7 days), stored in DB, only used to get new access tokens

  LOGIN:    server issues both → store refresh token in DB
  API CALL: client sends access token → server verifies (no DB hit)
  EXPIRE:   client sends refresh token → server checks DB → issues new access token
  BAN USER: delete refresh token from DB → next refresh attempt fails → locked out in ≤15min

  Invalidation speed: max 15 minutes
  DB lookups per request: 0 (except token refresh, rare)

INVALIDATION OPTIONS
--------------------
  Short expiry (15min)        → 0 DB lookups, max 15min exposure
  Access + refresh tokens     → 0 DB lookups per request, industry standard
  Redis blacklist             → instant invalidation, but 1 Redis hit per request

INVALID vs EXPIRED — CRITICAL DISTINCTION
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  ExpiredSignatureError  →  valid signature, exp timestamp passed
                            recoverable → tell client to refresh
  JWTError               →  bad signature, tampered, malformed
                            not recoverable → force re-login

  from jose import jwt, JWTError, ExpiredSignatureError

  def decode_token(token: str) -> dict:
      try:
          return jwt.decode(token, SECRET_KEY, algorithms=[ALGORITHM])
      except ExpiredSignatureError:
          raise HTTPException(401, "token expired, please refresh")
      except JWTError:
          raise HTTPException(401, "invalid token")

  Catching both as JWTError → client can't tell whether to refresh or re-login.
  Always distinguish them.

IMPLEMENTATION PATTERN
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  SECRET_KEY = os.environ["SECRET_KEY"]   # never hardcode
  ALGORITHM  = "HS256"

  def create_access_token(user_id: int) -> str:
      payload = {
          "user_id": user_id,
          "exp": datetime.utcnow() + timedelta(minutes=15),
          "type": "access"
      }
      return jwt.encode(payload, SECRET_KEY, algorithm=ALGORITHM)

  async def get_current_user(authorization: str = Header()):
      token = authorization[7:]   # strip "Bearer "
      payload = decode_token(token)
      return payload["user_id"]

  @app.get("/profile")
  async def profile(user_id: int = Depends(get_current_user)):
      return {"user_id": user_id}

SECURITY RULES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  SECRET_KEY must be long + random (32+ chars) — from env, never hardcoded
  Never put sensitive data in payload — it's base64, not encrypted
  Always set exp — tokens without expiry are valid forever
  Distinguish ExpiredSignatureError from JWTError — different client actions
  Check token type — access token ≠ refresh token, validate the "type" claim
